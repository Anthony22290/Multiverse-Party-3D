using System.Collections.Generic;

using UnityEngine;

// Five more first-person rooms. Same room space and wall conventions as EscapeRoomsFP.cs.

// ============================================================ lights out: the lantern wall
public sealed class RoomLanterns : EscapeRoom
{
    const int N = 4;
    const float Pitch = .8f;
    static readonly Vector3 Center = new(-5.4f, 2.1f, EscapeFP.ExitZ - .08f);
    readonly bool[] on = new bool[N * N], need = new bool[N * N];
    readonly Material[] glass = new Material[N * N];
    readonly EscapeInteractable[] uses = new EscapeInteractable[N * N];
    readonly float[] flash = new float[N * N];

    public override string Title => "Muro de los Faroles";
    public override string Objective => $"La puerta despierta cuando todos los faroles arden ({on.Count(x => x)}/{N * N}).";
    public override string[] Clues => new[]
    {
        "Ningún farol vive solo: lo que le haces a uno, también se lo haces a sus vecinos de arriba, abajo y los lados.",
        "Olvida el muro entero: deja perfecta la fila de arriba y luego, por cada farol apagado de una fila, pulsa el que tiene justo debajo. Lo que quede abajo te dirá qué tocar arriba."
    };
    public override Color Floor => new Color(.32f, .3f, .4f);
    public override Color Wall => new Color(.5f, .46f, .6f);
    public override Color Accent => new Color(1f, .7f, .3f);
    public override Color Glow => new Color(.75f, .7f, .95f);

    Vector3 Cell(int i) => Center + new Vector3((i % N - 1.5f) * Pitch, (1.5f - i / N) * Pitch, 0);

    public override void Build()
    {
        for (int i = 0; i < on.Length; i++) on[i] = true;
        int presses = Random.Range(6, 9);
        do
        {
            foreach (int i in Enumerable.Range(0, N * N).OrderBy(_ => Random.value).Take(presses)) Toggle(i, false);
        } while (on.All(x => x));

        Material wood = M("Madera oscura", new Color(.36f, .22f, .13f)), iron = M("Hierro", new Color(.3f, .32f, .36f));
        Block("Marco de los faroles", Center + new Vector3(0, 0, .05f), new Vector3(N * Pitch + .4f, N * Pitch + .4f, .1f), wood, .06f, false);
        for (int i = 0; i < N * N; i++)
        {
            var p = Cell(i);
            Block("Soporte", p + new Vector3(0, 0, -.02f), new Vector3(.5f, .58f, .12f), iron, .04f, false);
            glass[i] = EscapeKit.Glow(Accent, 0f);
            var lamp = EscapeKit.Shape(PrimitiveType.Sphere, transform, "Farol", p + new Vector3(0, 0, -.18f), new Vector3(.36f, .44f, .36f), glass[i], false);
            Block("Tapa", p + new Vector3(0, .26f, -.18f), new Vector3(.3f, .06f, .3f), iron, .02f, false);
            int idx = i;
            uses[i] = EscapeInteractable.Add(lamp, "Tocar el farol", 0f, a => Press(idx, a), a => Active, 2.6f);
            uses[i].maxBelow = 3.6f; uses[i].aimRadius = .28f;
        }
        EscapeKit.Text(transform, "QUE NINGUNO DUERMA", Center + new Vector3(0, -1.95f, -.1f), EscapeFP.OnExit, .022f, new Color(1f, .9f, .7f));
        EscapeKit.PointLight(transform, Center + new Vector3(0, 0, -2f), Accent, 6f, .6f);
        Prop("Banco", new Vector3(-6f, 0, 6.5f), 180); Prop("Maceta", new Vector3(8.6f, 0, -9.4f), 0);
    }

    void Toggle(int i, bool visual)
    {
        int x = i % N, y = i / N;
        need[i] = !need[i];
        foreach (var (dx, dy) in new[] { (0, 0), (1, 0), (-1, 0), (0, 1), (0, -1) })
        {
            int nx = x + dx, ny = y + dy;
            if (nx < 0 || ny < 0 || nx >= N || ny >= N) continue;
            int j = ny * N + nx; on[j] = !on[j];
            if (visual) flash[j] = 1f;
        }
    }

    void Press(int i, EscapeActor a)
    {
        Touch(a);
        Toggle(i, true);
        EscapeAudio.Play("tick", W(Cell(i)), .8f);
        if (on.All(x => x)) Solve(a);
    }

    public override void Tick()
    {
        for (int i = 0; i < N * N; i++)
        {
            flash[i] = Mathf.MoveTowards(flash[i], 0f, Time.deltaTime * 3f);
            bool lit = on[i] || Solved;
            EscapeKit.SetGlow(glass[i], lit ? Accent : new Color(.25f, .25f, .35f), (lit ? 2.4f + Mathf.Sin(Time.time * 7f + i) * .2f : .05f) + flash[i]);
        }
    }

    protected override void OnSolved() => Confetti(W(Center + new Vector3(0, 0, -.8f)), Accent, 70);

    public override EscapeOrder Order(EscapeActor a)
    {
        if (a.Slot != 1 || !Assist(120f)) return Wait(a);
        for (int i = 0; i < N * N; i++) if (need[i]) return EscapeOrder.Use(uses[i]);
        return Wait(a);
    }
}

// ============================================================ water jugs
public sealed class RoomJugs : EscapeRoom
{
    static readonly (int a, int b, int t)[] Sets = { (3, 5, 4), (4, 7, 5), (5, 8, 6), (3, 7, 5), (4, 9, 6) };
    int capA, capB, target, a, b;
    float shownA, shownB;
    Transform waterA, waterB;
    readonly Dictionary<string, EscapeInteractable> buttons = new();
    static readonly string[] Roman = { "", "I", "II", "III", "IV", "V", "VI", "VII", "VIII", "IX", "X" };
    static readonly Vector3 JugA = new(-3.2f, 0, 1.5f), JugB = new(3.2f, 0, 1.5f);
    const float JugHeight = 1.6f;

    public override string Title => "Fuente de las Medidas";
    public override string Objective => "El altar de la salida tiene sed, pero solo de una cantidad exacta.";
    public override string[] Clues => new[]
    {
        "Las jarras no tienen marcas. Solo sabes cuánto cabe en cada una cuando están llenas… o cuando una se llena con la otra.",
        "Llena una, viértela en la otra hasta que rebose, vacía la que se llenó y sigue vertiendo. Lo que sobra en el fondo es la medida que buscas."
    };
    public override Color Floor => new Color(.45f, .62f, .66f);
    public override Color Wall => new Color(.7f, .84f, .86f);
    public override Color Accent => new Color(.25f, .6f, .95f);
    public override Color Glow => new Color(.85f, .95f, 1f);

    public override void Build()
    {
        (capA, capB, target) = Sets[Random.Range(0, Sets.Length)];
        Material stone = M("Crema", new Color(.9f, .9f, .86f)), gold = M("Dorado", EscapeFP.Brass), wood = M("Madera oscura", new Color(.36f, .22f, .13f));
        var waterMat = EscapeKit.Glow(new Color(.3f, .65f, 1f), .5f);
        waterA = Jug(JugA, capA, stone, gold, waterMat, "A");
        waterB = Jug(JugB, capB, stone, gold, waterMat, "B");

        // Fountain between the jugs.
        Block("Fuente", new Vector3(0, .4f, 4.2f), new Vector3(2.2f, .8f, 1.4f), stone, .1f);
        EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Agua de la fuente", new Vector3(0, .82f, 4.2f), new Vector3(1.8f, .02f, 1f), waterMat, false);
        Block("Caño", new Vector3(0, 1.6f, 4.9f), new Vector3(.2f, 1.6f, .2f), gold, .04f, false);

        // Thirsty altar by the exit and its tablet (target in Roman numerals).
        Block("Altar", new Vector3(5.6f, .5f, 9.2f), new Vector3(1.6f, 1f, 1f), stone, .1f);
        EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Cuenco del altar", new Vector3(5.6f, 1.05f, 9.2f), new Vector3(1.1f, .08f, .7f), gold, false);
        Block("Tablilla", new Vector3(5.6f, 2.4f, EscapeFP.ExitZ - .04f), new Vector3(2.8f, 1f, .06f), M("Crema", EscapeFP.Parchment), .02f, false);
        EscapeKit.Text(transform, "EL ALTAR SOLO BEBE\n" + Roman[target] + " MEDIDAS", new Vector3(5.6f, 2.4f, EscapeFP.ExitZ - .08f), EscapeFP.OnExit, .028f, EscapeFP.Ink);
        Prop("Maceta", new Vector3(-8.6f, 0, -9.4f), 0); Prop("Banco", new Vector3(-8.6f, 0, 5f), 90);
    }

    Transform Jug(Vector3 at, int cap, Material stone, Material gold, Material waterMat, string id)
    {
        Block("Pedestal de la jarra " + id, at + new Vector3(0, .45f, 0), new Vector3(1.4f, .9f, 1.4f), stone, .1f);
        float h = JugHeight * cap / 9f + .4f, r = .9f;
        Vector3 bottom = at + new Vector3(0, .92f, 0);
        EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Fondo de la jarra", bottom, new Vector3(r, .03f, r), gold, false);
        for (int k = 0; k < 4; k++)
        {
            float ang = k * 90f * Mathf.Deg2Rad;
            Block("Varilla", bottom + new Vector3(Mathf.Cos(ang) * r * .48f, h * .5f, Mathf.Sin(ang) * r * .48f), new Vector3(.05f, h, .05f), gold, .01f, false);
        }
        EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Boca", bottom + new Vector3(0, h, 0), new Vector3(r + .05f, .02f, r + .05f), gold, false);
        EscapeKit.Text(transform, "CABEN " + Roman[cap], at + new Vector3(0, .62f, -.72f), EscapeFP.OnExit, .016f, EscapeFP.Ink);
        var water = new GameObject("Agua " + id).transform; water.SetParent(transform, false); water.localPosition = bottom;
        EscapeKit.Shape(PrimitiveType.Cylinder, water, "Agua", new Vector3(0, .5f, 0), new Vector3(r * .9f, .5f, r * .9f), waterMat, false);
        water.localScale = new Vector3(1, .001f, 1);
        water.gameObject.AddComponent<JugHeightTag>().height = h - .05f;
        // Three buttons on the pedestal front.
        bool isA = id == "A";
        string[] labels = { "LLENAR", "VACIAR", isA ? "VERTER →" : "← VERTER" };
        string[] keys = { "fill", "empty", "pour" };
        for (int i = 0; i < 3; i++)
        {
            Vector3 bp = at + new Vector3(-.45f + i * .45f, .55f, -.72f);
            var btn = EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Botón " + labels[i], bp, new Vector3(.24f, .04f, .24f), P(i == 0 ? new Color(.3f, .6f, 1f) : i == 1 ? new Color(.8f, .4f, .3f) : new Color(1f, .8f, .3f)), false, new Vector3(90, 0, 0));
            EscapeKit.Text(transform, labels[i], bp + new Vector3(0, .2f, -.02f), EscapeFP.OnExit, .0055f, EscapeFP.Ink);
            string key = id + keys[i];
            var use = EscapeInteractable.Add(btn, labels[i].Replace("→", "").Replace("←", "").Trim().ToLowerInvariant() switch
            {
                "llenar" => $"Llenar la jarra {id} en la fuente",
                "vaciar" => $"Vaciar la jarra {id}",
                _ => $"Verter la jarra {id} en la otra"
            }, 0f, act => Do(key, act), act => Active, 1.8f);
            use.aimRadius = .16f;
            buttons[key] = use;
        }
        return water;
    }

    (int, int) Apply(string key, int x, int y)
    {
        switch (key)
        {
            case "Afill": return (capA, y);
            case "Bfill": return (x, capB);
            case "Aempty": return (0, y);
            case "Bempty": return (x, 0);
            case "Apour": { int m = Mathf.Min(x, capB - y); return (x - m, y + m); }
            default: { int m = Mathf.Min(y, capA - x); return (x + m, y - m); }
        }
    }

    void Do(string key, EscapeActor act)
    {
        Touch(act);
        (a, b) = Apply(key, a, b);
        EscapeAudio.Play(key.EndsWith("empty") ? "fall" : "cell", W(key[0] == 'A' ? JugA : JugB) + Vector3.up, .5f);
        EscapeAudio.Play("plate_on", W(key[0] == 'A' ? JugA : JugB), .4f);
    }

    public override void Tick()
    {
        shownA = Mathf.MoveTowards(shownA, a, Time.deltaTime * 4f);
        shownB = Mathf.MoveTowards(shownB, b, Time.deltaTime * 4f);
        SetLevel(waterA, shownA / capA);
        SetLevel(waterB, shownB / capB);
        if (Active && (a == target || b == target) && Mathf.Abs(shownA - a) < .05f && Mathf.Abs(shownB - b) < .05f) Solve(lastTouch);
    }

    static void SetLevel(Transform w, float f) => w.localScale = new Vector3(1, Mathf.Max(.001f, f * w.GetComponent<JugHeightTag>().height), 1);

    protected override void OnSolved() { Confetti(W(new Vector3(5.6f, 1.6f, 9.2f)), Accent, 70); EscapeAudio.Play("chime", W(new Vector3(5.6f, 1.5f, 9.2f))); }

    string NextMove()
    {
        var start = (a, b);
        var prev = new Dictionary<(int, int), ((int, int) from, string key)> { [start] = (start, null) };
        var q = new Queue<(int, int)>(); q.Enqueue(start);
        (int, int)? goal = null;
        while (q.Count > 0)
        {
            var s = q.Dequeue();
            if (s.Item1 == target || s.Item2 == target) { goal = s; break; }
            foreach (var k in new[] { "Afill", "Bfill", "Aempty", "Bempty", "Apour", "Bpour" })
            {
                var n = Apply(k, s.Item1, s.Item2);
                if (prev.ContainsKey(n)) continue;
                prev[n] = (s, k); q.Enqueue(n);
            }
        }
        if (goal == null || goal.Value.Equals(start)) return null;
        var at = goal.Value;
        while (!prev[at].from.Equals(start)) at = prev[at].from;
        return prev[at].key;
    }

    public override EscapeOrder Order(EscapeActor act)
    {
        if (act.Slot != 1 || !Assist(120f)) return Wait(act);
        var k = NextMove();
        return k == null ? Wait(act) : EscapeOrder.Use(buttons[k]);
    }
}

public sealed class JugHeightTag : MonoBehaviour { public float height = 1f; }

// ============================================================ pipes
public sealed class RoomPipes : EscapeRoom
{
    const int Rows = 4, Cols = 5, Nb = 1, Eb = 2, Sb = 4, Wb = 8;
    const float Pitch = .78f;
    static readonly Vector3 Center = new(-5.4f, 2.15f, EscapeFP.ExitZ - .08f);
    sealed class Pipe { public int baseMask, rot, need = -1; public float shown; public Transform pivot; public Material mat; public EscapeInteractable use; }
    readonly Pipe[,] grid = new Pipe[Rows, Cols];
    readonly bool[,] wet = new bool[Rows, Cols];
    int r0, r1;
    bool flowing;
    float flowTime;
    Material drainMat;

    public override string Title => "Sala de las Tuberías";
    public override string Objective => flowing ? "¡El agua llega al molino!" : "El molino de la puerta necesita agua. Girad los tubos del muro.";
    public override string[] Clues => new[]
    {
        "El agua no salta huecos: cada tubo tiene que tocar al siguiente, boca con boca.",
        "Empieza en el grifo y avanza tubo a tubo; mira qué piezas ya se llenan de azul. Algunos tubos no llevan a ninguna parte."
    };
    public override Color Floor => new Color(.48f, .5f, .48f);
    public override Color Wall => new Color(.72f, .74f, .7f);
    public override Color Accent => new Color(.3f, .7f, 1f);
    public override Color Glow => new Color(.9f, .96f, 1f);

    static int Rot(int m) => ((m << 1) | (m >> 3)) & 15;
    static int MaskOf(Pipe p) { int m = p.baseMask; for (int i = 0; i < p.rot; i++) m = Rot(m); return m; }
    Vector3 CellPos(int r, int c) => Center + new Vector3((c - (Cols - 1) * .5f) * Pitch, ((Rows - 1) * .5f - r) * Pitch, 0);

    public override void Build()
    {
        // A path from the tap (left of row r0) to the drain (right of row r1).
        r0 = Random.Range(0, Rows); r1 = Random.Range(0, Rows);
        var cells = new List<(int r, int c)>();
        int row = r0;
        for (int c = 0; c < Cols; c++)
        {
            int to = c == Cols - 1 ? r1 : Random.Range(0, Rows);
            cells.Add((row, c));
            while (row != to) { row += to > row ? 1 : -1; cells.Add((row, c)); }
        }
        var need = new int[Rows, Cols];
        for (int i = 0; i < cells.Count; i++)
        {
            int m = 0;
            var (r, c) = cells[i];
            var prev = i == 0 ? (r, c - 1) : cells[i - 1];
            var next = i == cells.Count - 1 ? (r, c + 1) : cells[i + 1];
            foreach (var n in new[] { prev, next }) m |= Dir(r, c, n.Item1, n.Item2);
            need[r, c] = m;
        }
        Material frame = M("Madera oscura", new Color(.36f, .22f, .13f)), iron = M("Hierro", new Color(.3f, .32f, .36f));
        Block("Panel de tuberías", Center + new Vector3(0, 0, .05f), new Vector3(Cols * Pitch + .5f, Rows * Pitch + .4f, .1f), frame, .06f, false);
        for (int r = 0; r < Rows; r++)
            for (int c = 0; c < Cols; c++)
            {
                var p = new Pipe { need = need[r, c] };
                p.baseMask = p.need == 0 ? (Random.value < .5f ? Nb | Sb : Nb | Eb) : (p.need == (Nb | Sb) || p.need == (Eb | Wb) ? Nb | Sb : Nb | Eb);
                do p.rot = Random.Range(0, 4); while (p.need != 0 && MaskOf(p) == p.need);
                p.shown = p.rot * 90f;
                Block("Casilla", CellPos(r, c) + new Vector3(0, 0, -.01f), new Vector3(Pitch - .06f, Pitch - .06f, .04f), P(new Color(.22f, .2f, .2f)), .01f, false);
                var pivot = new GameObject("Tubo").transform; pivot.SetParent(transform, false); pivot.localPosition = CellPos(r, c) + new Vector3(0, 0, -.08f);
                p.mat = EscapeKit.Glow(new Color(.6f, .6f, .62f), 0f);
                EscapeKit.Block(pivot, "Codo", Vector3.zero, new Vector3(.2f, .2f, .1f), p.mat, .03f, false);
                if ((p.baseMask & Nb) != 0) EscapeKit.Block(pivot, "Tramo", new Vector3(0, Pitch * .25f, 0), new Vector3(.16f, Pitch * .5f, .08f), p.mat, .02f, false);
                if ((p.baseMask & Eb) != 0) EscapeKit.Block(pivot, "Tramo", new Vector3(Pitch * .25f, 0, 0), new Vector3(Pitch * .5f, .16f, .08f), p.mat, .02f, false);
                if ((p.baseMask & Sb) != 0) EscapeKit.Block(pivot, "Tramo", new Vector3(0, -Pitch * .25f, 0), new Vector3(.16f, Pitch * .5f, .08f), p.mat, .02f, false);
                p.pivot = pivot;
                var pp = p;
                p.use = EscapeInteractable.Add(pivot.gameObject, "Girar el tubo", 0f, a => { Touch(a); pp.rot = (pp.rot + 1) % 4; EscapeAudio.Play("tick", pp.pivot.position, .6f); }, a => Active && !flowing, 2.6f);
                p.use.maxBelow = 3.6f; p.use.aimRadius = .3f;
                grid[r, c] = p;
            }
        // Tap on the left, drain and water wheel on the right.
        var tap = CellPos(r0, 0) + new Vector3(-Pitch * .85f, 0, -.08f);
        Block("Grifo", tap, new Vector3(.5f, .26f, .2f), M("Dorado", EscapeFP.Brass), .05f, false);
        EscapeKit.Shape(PrimitiveType.Sphere, transform, "Rueda del grifo", tap + new Vector3(-.2f, .22f, -.05f), Vector3.one * .22f, P(new Color(.85f, .2f, .2f)), false);
        var drain = CellPos(r1, Cols - 1) + new Vector3(Pitch * .85f, 0, -.08f);
        drainMat = EscapeKit.Glow(Accent, .1f);
        Block("Desagüe", drain, new Vector3(.5f, .26f, .2f), drainMat, .05f, false);
        var wheel = new GameObject("Molino").transform; wheel.SetParent(transform, false); wheel.localPosition = new Vector3(-2.6f, 2.85f, EscapeFP.ExitZ - .3f);
        for (int k = 0; k < 6; k++) EscapeKit.Block(wheel, "Pala", Vector3.zero, new Vector3(.12f, 1.4f, .1f), iron, .02f, false, new Vector3(0, 0, k * 30f));
        var spin = wheel.gameObject.AddComponent<EscapeSpin>(); spin.axis = Vector3.forward; spin.speed = 0f;
        wheelSpin = spin;
        Prop("Cajas", new Vector3(8.4f, 0, -8.6f), 20); Prop("Maceta", new Vector3(-8.6f, 0, -9.4f), 0);
    }

    EscapeSpin wheelSpin;

    static int Dir(int r, int c, int nr, int nc) => nr < r ? Nb : nr > r ? Sb : nc > c ? Eb : Wb;

    public override void Tick()
    {
        foreach (var p in grid)
        {
            p.shown = Mathf.MoveTowardsAngle(p.shown, p.rot * 90f, Time.deltaTime * 400f);
            p.pivot.localRotation = Quaternion.Euler(0, 0, -p.shown);
        }
        // Flood from the tap.
        System.Array.Clear(wet, 0, wet.Length);
        bool reached = false;
        if ((MaskOf(grid[r0, 0]) & Wb) != 0)
        {
            var q = new Queue<(int, int)>(); q.Enqueue((r0, 0)); wet[r0, 0] = true;
            while (q.Count > 0)
            {
                var (r, c) = q.Dequeue();
                int m = MaskOf(grid[r, c]);
                if (r == r1 && c == Cols - 1 && (m & Eb) != 0) reached = true;
                foreach (var (bit, dr, dc, back) in new[] { (Nb, -1, 0, Sb), (Sb, 1, 0, Nb), (Eb, 0, 1, Wb), (Wb, 0, -1, Eb) })
                {
                    if ((m & bit) == 0) continue;
                    int nr = r + dr, nc = c + dc;
                    if (nr < 0 || nc < 0 || nr >= Rows || nc >= Cols || wet[nr, nc]) continue;
                    if ((MaskOf(grid[nr, nc]) & back) == 0) continue;
                    wet[nr, nc] = true; q.Enqueue((nr, nc));
                }
            }
        }
        for (int r = 0; r < Rows; r++) for (int c = 0; c < Cols; c++)
            EscapeKit.SetGlow(grid[r, c].mat, wet[r, c] ? Accent : new Color(.6f, .6f, .62f), wet[r, c] ? 1.6f : 0f);
        EscapeKit.SetGlow(drainMat, Accent, reached ? 2.4f : .1f);
        if (reached && Active && Mathf.Abs(Mathf.DeltaAngle(grid[r1, Cols - 1].shown, grid[r1, Cols - 1].rot * 90f)) < 1f)
        {
            flowing = true; flowTime += Time.deltaTime;
            if (flowTime > .8f) Solve(lastTouch);
        }
        wheelSpin.speed = Mathf.MoveTowards(wheelSpin.speed, reached || Solved ? 120f : 0f, Time.deltaTime * 60f);
    }

    protected override void OnSolved() { Confetti(W(new Vector3(-2.6f, 2.8f, 10f)), Accent, 70); EscapeAudio.Play("chime", W(new Vector3(-2.6f, 2.8f, 10f))); }

    public override EscapeOrder Order(EscapeActor a)
    {
        if (a.Slot != 1 || !Assist(120f)) return Wait(a);
        foreach (var p in grid)
            if (p.need != 0 && MaskOf(p) != p.need && Mathf.Abs(Mathf.DeltaAngle(p.shown, p.rot * 90f)) < 2f) return EscapeOrder.Use(p.use);
        return Wait(a);
    }
}

// ============================================================ number sequences (one on the ceiling)
public sealed class RoomNumbers : EscapeRoom
{
    readonly int[] answers = new int[3];
    string code;
    EscapeInteractable keypad;
    float lockedUntil;
    Material screen;

    public override string Title => "Observatorio de los Números";
    public override string Objective => "El teclado de la salida pide un código. Las estrellas del observatorio lo saben.";
    public override string[] Clues => new[]
    {
        "Hay más números en esta sala de los que se ven de frente. Los astrónomos miran hacia arriba.",
        "El teclado pide primero la respuesta de la serie I y después la de la II (dos cifras cada una). La serie III es un engaño."
    };
    public override Color Floor => new Color(.18f, .2f, .34f);
    public override Color Wall => new Color(.32f, .34f, .52f);
    public override Color Accent => new Color(.5f, .7f, 1f);
    public override Color Glow => new Color(.7f, .8f, 1f);

    static int[] Sequence(out int answer)
    {
        while (true)
        {
            var t = new int[6];
            switch (Random.Range(0, 4))
            {
                case 0: { int s = Random.Range(2, 15), d = Random.Range(3, 13); for (int i = 0; i < 6; i++) t[i] = s + i * d; break; }
                case 1: { int s = Random.Range(1, 5), k = Random.Range(0, 12); for (int i = 0; i < 6; i++) t[i] = (s + i) * (s + i) + k; break; }
                case 2: { t[0] = Random.Range(1, 8); t[1] = Random.Range(t[0], 10); for (int i = 2; i < 6; i++) t[i] = t[i - 1] + t[i - 2]; break; }
                default: { int s = Random.Range(1, 20), d = Random.Range(1, 6); t[0] = s; for (int i = 1; i < 6; i++) t[i] = t[i - 1] + d + i - 1; break; }
            }
            if (t[5] >= 10 && t[5] <= 99 && t[4] < t[5]) { answer = t[5]; return t; }
        }
    }

    static string Show(int[] t) => string.Join("  ·  ", t.Take(5)) + "  ·  ?";

    public override void Build()
    {
        var seqs = new int[3][];
        for (int i = 0; i < 3; i++) seqs[i] = Sequence(out answers[i]);
        code = answers[0].ToString("00") + answers[1].ToString("00");
        Material dark = M("Azul noche", new Color(.1f, .12f, .25f)), gold = M("Dorado", EscapeFP.Brass);
        Color ink = new Color(1f, .93f, .7f);

        // I: on the ceiling, above the middle of the room (look up!).
        var ceil = new Vector3(0, EscapeLevel.WallHeight - .05f, -2.5f);
        Block("Placa del techo", ceil, new Vector3(6.4f, .08f, 2.2f), dark, .02f, false);
        EscapeKit.Text(transform, "I", ceil + new Vector3(0, -.06f, -.75f), new Vector3(-90, 0, 0), .05f, EscapeFP.Brass);
        EscapeKit.Text(transform, Show(seqs[0]), ceil + new Vector3(0, -.06f, .15f), new Vector3(-90, 0, 0), .028f, ink);
        var starMat = EscapeKit.Glow(new Color(.8f, .9f, 1f), 1.8f);
        for (int k = 0; k < 30; k++)
            EscapeKit.Shape(PrimitiveType.Sphere, transform, "Estrella del techo", new Vector3(Random.Range(-9f, 9f), EscapeLevel.WallHeight - .02f, Random.Range(-10f, 10f)), Vector3.one * Random.Range(.05f, .12f), starMat, false);

        // II: low on the left wall, behind a column.
        var low = new Vector3(-EscapeFP.SideX + .05f, .75f, 3.5f);
        Block("Placa baja", low, new Vector3(.06f, .9f, 3.6f), dark, .02f, false);
        EscapeKit.Text(transform, "II", low + new Vector3(.06f, .28f, 0), EscapeFP.OnLeft, .03f, EscapeFP.Brass);
        EscapeKit.Text(transform, Show(seqs[1]), low + new Vector3(.06f, -.12f, 0), EscapeFP.OnLeft, .02f, ink);
        Block("Columna", new Vector3(-8.4f, EscapeLevel.WallHeight * .5f, 3.5f), new Vector3(1f, EscapeLevel.WallHeight, 1.4f), M("Crema", new Color(.8f, .8f, .86f)), .1f);

        // III: big and obvious on the right wall (the decoy).
        var deco = new Vector3(EscapeFP.SideX - .05f, 2.2f, 0f);
        Block("Placa grande", deco, new Vector3(.06f, 1.4f, 5f), dark, .02f, false);
        EscapeKit.Text(transform, "III", deco + new Vector3(-.06f, .42f, 0), EscapeFP.OnRight, .04f, EscapeFP.Brass);
        EscapeKit.Text(transform, Show(seqs[2]), deco + new Vector3(-.06f, -.12f, 0), EscapeFP.OnRight, .032f, ink);

        // Telescope (decor) and the keypad.
        Block("Trípode", new Vector3(5.5f, .8f, -6f), new Vector3(.3f, 1.6f, .3f), gold, .04f);
        Block("Telescopio", new Vector3(5.5f, 1.9f, -5.8f), new Vector3(.36f, .36f, 1.8f), M("Hierro", new Color(.3f, .32f, .36f)), .06f, false, new Vector3(-35, 20, 0));
        var pad = new Vector3(3.2f, 0, EscapeFP.ExitZ - 1.8f);
        Block("Atril del teclado", pad + new Vector3(0, .55f, 0), new Vector3(.9f, 1.1f, .6f), M("Hierro", new Color(.3f, .32f, .36f)), .08f);
        screen = EscapeKit.Glow(new Color(.4f, .8f, 1f), .8f);
        Block("Pantalla", pad + new Vector3(0, 1.2f, -.1f), new Vector3(.8f, .46f, .08f), screen, .03f, false, new Vector3(-30, 0, 0));
        EscapeKit.Text(transform, "I  ·  II", pad + new Vector3(0, 1.62f, -.2f), EscapeFP.OnExit, .02f, ink);
        var kp = new GameObject("Teclado"); kp.transform.SetParent(transform, false); kp.transform.localPosition = pad + new Vector3(0, 1.1f, -.3f);
        keypad = EscapeInteractable.Add(kp, "Escribir el código", 0f, UseKeypad, a => Active && Time.time >= lockedUntil, 2f);
    }

    void UseKeypad(EscapeActor a)
    {
        Touch(a);
        if (a.DrivenByBot) { Submit(Assist(110f) ? code : (answers[2] * 100 + answers[0]).ToString("0000"), a); return; }
        D.HUD.OpenKeypad("CÓDIGO DEL OBSERVATORIO", c => Submit(c, a));
    }

    bool Submit(string guess, EscapeActor a)
    {
        EscapeAudio.Play("beep", keypad.Point);
        if (guess == code) { EscapeKit.SetGlow(screen, new Color(.4f, 1f, .5f), 3f); Solve(a); return true; }
        lockedUntil = Time.time + 4f;
        EscapeKit.SetGlow(screen, new Color(1f, .3f, .3f), 2.5f);
        EscapeAudio.Play("error", keypad.Point);
        D.Notify("Código incorrecto · teclado bloqueado 4 s");
        return false;
    }

    public override void Tick()
    {
        if (!Solved && Time.time > lockedUntil) EscapeKit.SetGlow(screen, new Color(.4f, .8f, 1f), .8f + Mathf.Sin(Time.time * 3f) * .2f);
    }

    protected override void OnSolved() => Confetti(W(new Vector3(0, 2.5f, -2.5f)), Accent, 70);

    public override EscapeOrder Order(EscapeActor a)
    {
        if (a.Slot != 1 || !Assist(110f) || Time.time < lockedUntil) return Wait(a);
        return EscapeOrder.Use(keypad);
    }
}

// ============================================================ hidden footprints (lantern)
public sealed class RoomFootprints : EscapeRoom
{
    const int Rows = 6, Cols = 8;
    const float PitchX = 2.45f, PitchZ = 2.1f, MinX = -9.8f, MinZ = -6.3f;
    readonly bool[,] safe = new bool[Rows, Cols];
    readonly List<(int r, int c)> path = new();
    readonly List<Renderer> prints = new();
    readonly Dictionary<EscapeActor, int> progress = new();
    EscapeItem lantern;
    float trapCooldown;

    public override string Title => "Pasillo de las Huellas";
    public override string Objective => Solved ? "Las trampas duermen: cruzad todos." : "El suelo está lleno de trampas. Alguien cruzó antes que vosotros…";
    public override string[] Clues => new[]
    {
        "Los que cruzaron antes dejaron huellas, pero solo se ven con la luz adecuada.",
        "Coge el farol del pedestal de la entrada y avanza despacio: pisa solo las losas donde brillan las huellas. Quien lo lleva puede guiar a los demás."
    };
    public override Color Floor => new Color(.5f, .44f, .36f);
    public override Color Wall => new Color(.68f, .6f, .5f);
    public override Color Accent => new Color(.4f, 1f, .85f);
    public override Color Glow => new Color(.45f, .42f, .5f);

    static float X(int c) => MinX + (c + .5f) * PitchX;
    static float Z(int r) => MinZ + (r + .5f) * PitchZ;
    protected override Vector3 WaitSpot(int slot) => new Vector3(-6f + slot * 2f, 0f, 8.6f);

    public override void Build()
    {
        int col = Random.Range(0, Cols);
        for (int r = 0; r < Rows; r++)
        {
            if (!safe[r, col]) { safe[r, col] = true; path.Add((r, col)); }
            if (r < Rows - 1 && Random.value < .75f)
            {
                int target = Mathf.Clamp(col + (Random.value < .5f ? -1 : 1) * Random.Range(1, 4), 0, Cols - 1);
                while (col != target) { col += target > col ? 1 : -1; safe[r, col] = true; path.Add((r, col)); }
            }
        }
        Material a = P(new Color(.62f, .56f, .48f)), b = P(new Color(.56f, .5f, .43f));
        var printMat = EscapeKit.Glow(Accent, 2.2f);
        for (int r = 0; r < Rows; r++)
            for (int c = 0; c < Cols; c++)
            {
                Block("Losa", new Vector3(X(c), .015f, Z(r)), new Vector3(PitchX - .08f, .03f, PitchZ - .08f), (r + c) % 2 == 0 ? a : b, .01f, false);
                if (!safe[r, c]) continue;
                for (int f = -1; f <= 1; f += 2)
                {
                    var foot = EscapeKit.Shape(PrimitiveType.Sphere, transform, "Huella", new Vector3(X(c) + f * .22f, .035f, Z(r) + f * .25f), new Vector3(.2f, .01f, .34f), printMat, false);
                    var rend = foot.GetComponent<Renderer>(); rend.enabled = false; prints.Add(rend);
                }
            }
        Material gold = M("Dorado", EscapeFP.Brass);
        Block("Borde de salida", new Vector3(0, .03f, MinZ - .1f), new Vector3(EscapeRoom.Width, .05f, .12f), gold, .02f, false);
        Block("Borde de llegada", new Vector3(0, .03f, MinZ + Rows * PitchZ + .1f), new Vector3(EscapeRoom.Width, .05f, .12f), gold, .02f, false);
        // Dart holes on the walls (decor) and the lantern pedestal.
        for (int r = 0; r < Rows; r++) for (int s = -1; s <= 1; s += 2)
            EscapeKit.Shape(PrimitiveType.Cylinder, transform, "Agujero de dardo", new Vector3(s * (EscapeFP.SideX - .02f), .9f, Z(r)), new Vector3(.12f, .02f, .12f), P(new Color(.1f, .08f, .06f)), false, new Vector3(0, 0, 90));
        var ped = new Vector3(-6.5f, 0, -9.3f);
        Block("Pedestal del farol", ped + new Vector3(0, .5f, 0), new Vector3(.7f, 1f, .7f), M("Crema", new Color(.9f, .86f, .8f)), .08f);
        lantern = EscapeItem.Create(transform, "farol", "el farol de las huellas", ped + new Vector3(0, 1.35f, 0), it =>
        {
            EscapeKit.Block(it.transform, "Cuerpo", Vector3.zero, new Vector3(.26f, .34f, .26f), EscapeKit.Glow(Accent, 2.4f), .05f, false);
            EscapeKit.Block(it.transform, "Tapa", new Vector3(0, .21f, 0), new Vector3(.3f, .06f, .3f), gold, .02f, false);
            EscapeKit.PointLight(it.transform, Vector3.zero, Accent, 4f, 1.2f);
        });
        EscapeKit.Text(transform, "PISA DONDE PISARON", new Vector3(0, .05f, -8f), new Vector3(90, 0, 0), .04f, new Color(.95f, .9f, .8f));
        Prop("Antorcha tiki", new Vector3(-8.9f, 0, 9.4f), 0); Prop("Antorcha tiki", new Vector3(8.9f, 0, 9.4f), 0);
    }

    public override void Tick()
    {
        Vector3 lp = lantern != null ? lantern.transform.position : Vector3.one * 9999f;
        foreach (var r in prints)
        {
            Vector3 d = r.transform.position - lp; d.y = 0f;
            r.enabled = Solved || d.magnitude < 3.4f;
        }
        if (!Active) return;
        trapCooldown -= Time.deltaTime;
        foreach (var a in Team)
        {
            if (!InRoom(a) || !a.Grounded) continue;
            var p = L(a.transform.position);
            if (p.z > MinZ + Rows * PitchZ + .25f) { Solve(a); return; }
            int c = Mathf.FloorToInt((p.x - MinX) / PitchX), r = Mathf.FloorToInt((p.z - MinZ) / PitchZ);
            if (r < 0 || r >= Rows || c < 0 || c >= Cols || safe[r, c]) continue;
            EscapeAudio.Play("zap", a.transform.position, .9f);
            a.Knock(W(new Vector3(Mathf.Clamp(p.x, -8f, 8f), .05f, -8.4f)), .7f);
            progress.Remove(a);
            if (a.IsHuman || trapCooldown <= 0f) { D.Notify(a.IsHuman ? "¡Trampa! Vuelves a la entrada" : "¡" + a.DisplayName + " ha pisado una trampa!"); trapCooldown = 2f; }
        }
    }

    protected override void OnSolved() => Confetti(W(new Vector3(0, 1.5f, 8f)), Accent, 70);

    public override EscapeOrder Order(EscapeActor a)
    {
        if (Solved) return Wait(a);
        Vector3 lp = L(a.transform.position);
        bool walker = a.Slot == 1 && Assist(110f);
        if (!walker) return EscapeOrder.Go(W(new Vector3(-6f + a.Slot * 2.2f, 0, -8.6f)), .5f);
        if (!progress.TryGetValue(a, out int i) || lp.z < MinZ - .3f) i = -1;
        if (i == -1)
        {
            var approach = W(new Vector3(X(path[0].c), 0, MinZ - .7f));
            Vector3 d = a.transform.position - approach; d.y = 0f;
            if (d.magnitude > .3f) { progress[a] = -1; return EscapeOrder.Exact(approach); }
            i = 0;
        }
        if (i < path.Count)
        {
            Vector3 c = W(new Vector3(X(path[i].c), 0, Z(path[i].r)));
            Vector3 d = a.transform.position - c; d.y = 0f;
            if (d.magnitude < .3f) i++;
        }
        progress[a] = i;
        if (i >= path.Count) return EscapeOrder.Exact(W(new Vector3(X(path[path.Count - 1].c), 0, MinZ + Rows * PitchZ + 1f)));
        return EscapeOrder.Exact(W(new Vector3(X(path[i].c), 0, Z(path[i].r))));
    }
}
